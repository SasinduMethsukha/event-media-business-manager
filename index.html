SECURITY DEFINER avoids recursive profile RLS. create or replace function public.app_workspace_id() returns text language sql stable security definer set search_path = public as $$ select p.workspace_id from public.app_profiles p where p.user_id = auth.uid() and p.active = true limit 1 $$; create or replace function public.app_role() returns text language sql stable security definer set search_path = public as $$ select p.role from public.app_profiles p where p.user_id = auth.uid() and p.active = true limit 1 $$; create or replace function public.app_is_admin() returns boolean language sql stable security definer set search_path = public as $$ select coalesce(public.app_role() = 'admin', false) $$; grant execute on function public.app_workspace_id() to authenticated; grant execute on function public.app_role() to authenticated; grant execute on function public.app_is_admin() to authenticated; -- New Auth users are converted into app profiles. -- First profile for a workspace becomes Admin. -- Later employee sign-ups require an unused invite code generated by the Admin. create or replace function public.handle_event_media_new_user() returns trigger language plpgsql security definer set search_path = public as $$ declare v_ws text; v_name text; v_invite text; v_is_first boolean; v_invite_id uuid; begin v_ws := nullif(trim(coalesce(new.raw_user_meta_data->>'workspace_id','')), ''); v_name := trim(coalesce(new.raw_user_meta_data->>'full_name','')); v_invite := nullif(trim(coalesce(new.raw_user_meta_data->>'invite_code','')), ''); if v_ws is null then return new; end if; select not exists(select 1 from public.app_profiles where workspace_id = v_ws) into v_is_first; if v_is_first then insert into public.app_profiles(user_id,workspace_id,email,full_name,role,active) values(new.id,v_ws,new.email,v_name,'admin',true) on conflict(user_id) do nothing; return new; end if; if v_invite is not null then select id into v_invite_id from public.employee_invites where workspace_id = v_ws and invite_code = v_invite and used_at is null and expires_at > now() limit 1; end if; if v_invite_id is not null then insert into public.app_profiles(user_id,workspace_id,email,full_name,role,active) values(new.id,v_ws,new.email,v_name,'employee',true) on conflict(user_id) do nothing; update public.employee_invites set used_at = now() where id = v_invite_id; else insert into public.app_profiles(user_id,workspace_id,email,full_name,role,active) values(new.id,v_ws,new.email,v_name,'employee',false) on conflict(user_id) do nothing; end if; return new; end; $$; drop trigger if exists on_event_media_user_created on auth.users; create trigger on_event_media_user_created after insert on auth.users for each row execute function public.handle_event_media_new_user();
